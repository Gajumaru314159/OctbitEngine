298 constexpr Quat& Quat::operator *= (const Quat& quat) noexcept {
299 f32 x = this->x;
300 f32 y = this->y;
301 f32 z = this->z;
302 f32 w = this->w;
303
304 this->x = y * quat.z - z * quat.y + x * quat.w + w * quat.x;
305 this->y = z * quat.x - x * quat.z + y * quat.w + w * quat.y;
306 this->z = x * quat.y - y * quat.x + z * quat.w + w * quat.z;
307 this->w = w * quat.w - x * quat.x - y * quat.y - z * quat.z;
308 return *this;
309 }
