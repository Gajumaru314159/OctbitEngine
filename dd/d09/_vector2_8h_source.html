773 inline Vec2 Vec2::Slerp(const Vec2& a, const Vec2& b, f32 t) {
774 f32 al, bl;
775 Vec2 an, bn;
776 a.toDirectionAndLength(an, al);
777 b.toDirectionAndLength(bn, bl);
778 f32 angle = Math::Acos(Dot(an, bn));
779 f32 ps = Math::Sin(angle * (1.0f - t)) * Math::Lerp(al, bl, t);
780 f32 pe = Math::Sin(angle * t) * Math::Lerp(al, bl, t);
781
782 return Vec2(
783 an.x * ps + bn.x * pe,
784 an.y * ps + bn.y * pe
785 );
786 }
