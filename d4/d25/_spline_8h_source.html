44 constexpr T Spline::Hermite(const T& p0, const T& p1, const T& m0, const T& m1, f32 t) noexcept {
45 return (2 * (t * t * t) - 3 * (t * t) + 1) * p0 + ((t * t * t) - 2 * (t * t) + t) * m0 + ((t * t * t) - (t * t)) * m1 + (-2 * (t * t * t) + 3 * (t * t)) * p1;
46 }

58 constexpr T Spline::CalculateTangent(const T& p0, const T& p1, const T& p2, f32 a, f32 b) noexcept {
