256 // @brief 高速余弦正弦関数
257 inline void Math::SinCos(f32 radians, f32& outSin, f32& outCos)noexcept {
258 radians = Math::Wrap(radians, -PI, PI);
259
260 static f32 waru[8] = { -1.0f / (3 * 4 * 5 * 6 * 7 * 8),-1.0f / (2 * 3 * 4 * 5 * 6 * 7),1.0f / (3 * 4 * 5 * 6),1.0f / (2 * 3 * 4 * 5),-1.0f / (3 * 4),-1.0f / (2 * 3),1.0f,1.0f };
261 f32 c = 1.0f / (3 * 4 * 5 * 6 * 7 * 8 * 9 * 10);
262 f32 s = 1.0f / (2 * 3 * 4 * 5 * 6 * 7 * 8 * 9);
263 f32* p = waru;
264 f32 z;
265
266 s32 i;
267 radians *= (1.0f / 32.0f);
268 z = radians * radians;
269 do {
270 c = c * z + (*p);
271 p++;
272 s = s * z + (*p);
273 p++;
274 } while (p < waru + 8);
275 c = c * z;
276 s = s * radians;
277 for (i = 0; i < 5; i++) {
278 s = s * (2.0f - c);
279 c = c * (4.0f - c);
280 }
281 outSin = s;
282 outCos = 1.0f - c / 2.0f;
283 }
