486 auto _method_impl(StringView name, M method, Names&&... argNames)
487 -> std::enable_if_t<MethodTraits<M>::Count == sizeof...(Names) || 0 == sizeof...(Names), TagBuilder >
488 {
489 OB_ASSERT(!m_info.methods.contains(name), "{}は登録済みのメソッドです [{}]", name, m_info.type.name());
490
491 m_info.methodOrder.emplace_back(name);
492 auto& info = m_info.methods[name];
493 info.name = name;
494 info.returnType = Type::Get<typename MethodTraits<M>::return_type>();
495 info.isConst = MethodTraits<M>::Const;
496
497 // 0引数(引数名未指定)に対応するために最後尾に空要素を追加している
498 Array<StringView,sizeof...(Args)+1> names = {StringView(argNames)... ,""};
499 auto types = MethodTraits<M>::Types();
500
501 for (s32 i = 0; i + 1 < std::size(types); ++i) {
502 auto& arg = info.arguments.emplace_back();
503 arg.type = types.at(i);
504 arg.name = (sizeof...(Names) == 0) ? _GetDefaultArgumentName(i) : names.at(i);
505 }
506
507 if constexpr (sizeof...(Args) == 0) {
508 info.invoke = [=](Any& owner, Span<Any> args) { return _InvokeWithoutArgs<M, Args...>(owner, args, method); };
509 }
510 else {
511 info.invoke = [=](Any& owner, Span<Any> args) { return _InvokeMethod<M, Args...>(owner, args, method); };
512 }
513
514 return info;

613 return {};
