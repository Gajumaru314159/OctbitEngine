479 constexpr Color Color::AlphaBlend(const Color& dst, const Color& src) noexcept {
480 auto a = src.a + dst.a * (1 - src.a);
481 auto rgb = (src * src.a + dst * dst.a * (1 - src.a)) / a;
482 if (Math::IsNearZero(a))rgb = Color(0);
483 return {rgb.r, rgb.g, rgb.a, a};
484 }

495 constexpr Color Color::PremultipliedAlphaBlend(const Color& dst, const Color& src) noexcept {
496 auto a = src.a + dst.a * (1 - src.a);
497 auto rgb = src + dst * (1 - src.a);
498 return {rgb.r, rgb.g, rgb.a, a};
