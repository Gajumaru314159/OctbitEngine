480 constexpr Color Color::AlphaBlend(const Color& dst, const Color& src) noexcept {
481 auto a = src.a + dst.a * (1 - src.a);
482 auto rgb = (src * src.a + dst * dst.a * (1 - src.a)) / a;
483 if (Math::IsNearZero(a))rgb = Color(0);
484 return Color(rgb.r, rgb.g, rgb.a, a);
485 }

496 constexpr Color Color::PremultipliedAlphaBlend(const Color& dst, const Color& src) noexcept {
497 auto a = src.a + dst.a * (1 - src.a);
498 auto rgb = src + dst * (1 - src.a);
499 return Color(rgb.r, rgb.g, rgb.a, a);
