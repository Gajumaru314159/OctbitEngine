158 inline Vec3 Triangle::innerGravity()const noexcept {
159 f32 a = Vec3::Dist(p1, p2);
160 f32 b = Vec3::Dist(p2, p0);
161 f32 c = Vec3::Dist(p0, p1);
162 if (a + b + c < Math::EPSILON)return p0;
163 return (p0 * a + p1 * b + p2 * c) / (a + b + c);
164 }

168 inline Vec3 Triangle::circumcenter()const noexcept {
169 auto a = Vec3::SqrDist(p1, p2);
170 auto b = Vec3::SqrDist(p2, p0);
171 auto c = Vec3::SqrDist(p0, p1);
172 auto t_a = a * (-a + b + c);
173 auto t_b = b * (a - b + c);
174 auto t_c = c * (a + b - c);
175 auto s = t_a + t_b + t_c;
176 return t_a / s * p0 + t_b / s*p1 +t_c / s * p2;
177 }

181 inline f32 Triangle::area()const noexcept {

183 return Math::Sqrt(Vec3::SqrDist(p0, p1) * Vec3::SqrDist(p0, p2) - abac * abac) * 0.5f;
184 }
