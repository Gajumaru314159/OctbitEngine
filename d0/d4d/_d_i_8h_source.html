324 template<class T>
325 void* ServiceBuilder<T>::create(ServiceContainer& container) {
326 // 生成
327 detail::ServiceHolder<T>* holder = nullptr;
328 if (m_getter) {
329 holder = new detail::ServiceHolder<T>(m_getter(), false);
330 }
331 else {
332 holder = new detail::ServiceHolder<T>(reinterpret_cast<T*>(detail::Factory<T>::Create(m_injector, container)), true);
333 }
334 // 無効
335 if (holder->get() == nullptr) {
336 delete holder;
337 return nullptr;
338 }
339 // 基底クラスをマッピング
340 auto index = container.m_services.size();
341 for (auto& type : m_bases) {
342 container.m_indices.emplace(type, index);
343 }
344 return container.m_services.emplace_back(holder)->get();
345 }

348 template<class T>
349 T* ServiceInjector::create(ServiceContainer& container)const {
350 // 生成済み
351 if (auto instance = container.get<T>())
352 return instance;
353 // 抽象->具象
354 Vector<Type> fallback;
355 const auto& concretes = try_find(m_builderMap, Type::Get<T>(), fallback);
356 // 生成
357 for (auto& concrete : concretes) {
358 auto& builder = m_builders.find(concrete)->second;
359 try {
360 return reinterpret_cast<T*>(builder->create(container));
361 }
362 catch (const std::exception& e) {
363 LOG_TRACE("[DI] {}の生成がキャンセルされました。\n{}", Type::Get<T>().name(), e.what());
364 }
365 }
366 return nullptr;
367 }
