template<class TKey, class TValue, class THasher = std::hash<TKey>, class TKeyEqual = std::equal_to<TKey>, class TAllocator = ob::core::STLAllocator<Pair<TKey, TValue>>>
class ob::core::IndexMap< TKey, TValue, THasher, TKeyEqual, TAllocator >

安定した反復順序を持つ順序付きハッシュマップ

VectorストレージとHashMapインデックスにより挿入順序を維持しながら、平均O(1)の検索/更新を実現します。 内部のキーとインデックスのマッピングが壊れないようにするため、キーは挿入後も不変である必要があります。
