1020 inline Vec3 Vec3::Slerp(const Vec3& a, const Vec3& b, f32 t) {
1021 f32 al, bl;
1022 Vec3 an, bn;
1023 a.toDirectionAndLength(an, al);
1024 b.toDirectionAndLength(bn, bl);
1025 f32 angle = Math::Acos(Dot(an, bn));
1026 f32 ps = Math::Sin(angle * (1.0f - t)) * Math::Lerp(al, bl, t);
1027 f32 pe = Math::Sin(angle * t) * Math::Lerp(al, bl, t);
1028
1029 return Vec3(
1030 an.x * ps + bn.x * pe,
1031 an.y * ps + bn.y * pe,
1032 an.z * ps + bn.z * pe
1033 );
1034 }
