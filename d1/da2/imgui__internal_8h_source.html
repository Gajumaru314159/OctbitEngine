702// Helper: ImChunkStream<>
703// Build and iterate a contiguous stream of variable-sized structures.
704// This is used by Settings to store persistent data while reducing allocation count.
705// We store the chunk size first, and align the final size on 4 bytes boundaries.
706// The tedious/zealous amount of casting is to avoid -Wcast-align warnings.
707template<typename T>
708struct ImChunkStream
709{
710 ImVector<char> Buf;
711
712 void clear() { Buf.clear(); }
713 bool empty() const { return Buf.Size == 0; }
714 int size() const { return Buf.Size; }
715 T* alloc_chunk(size_t sz) { size_t HDR_SZ = 4; sz = IM_MEMALIGN(HDR_SZ + sz, 4u); int off = Buf.Size; Buf.resize(off + (int)sz); ((int*)(void*)(Buf.Data + off))[0] = (int)sz; return (T*)(void*)(Buf.Data + off + (int)HDR_SZ); }
716 T* begin() { size_t HDR_SZ = 4; if (!Buf.Data) return NULL; return (T*)(void*)(Buf.Data + HDR_SZ); }

743// ImDrawList: Helper function to calculate a circle's segment count given its radius and a "maximum error" value.
744// Estimation of number of circle segment based on error is derived using method described in https://stackoverflow.com/a/2244088/15194693
745// Number of segments (N) is calculated using equation:
746// N = ceil ( pi / acos(1 - error / r) ) where r > 0, error <= r
747// Our equation is significantly simpler that one in the post thanks for choosing segment that is
748// perpendicular to X axis. Follow steps in the article from this starting condition and you will
749// will get this result.
750//
751// Rendering circles with an odd number of segments, while mathematically correct will produce
752// asymmetrical results on the raster grid. Therefore we're rounding N to next even number (7->8, 8->8, 9->10 etc.)
753#define IM_ROUNDUP_TO_EVEN(_V) ((((_V) + 1) / 2) * 2)
754#define IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_MIN 4
755#define IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_MAX 512
756#define IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_CALC(_RAD,_MAXERROR) ImClamp(IM_ROUNDUP_TO_EVEN((int)ImCeil(IM_PI / ImAcos(1 - ImMin((_MAXERROR), (_RAD)) / (_RAD)))), IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_MIN, IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_MAX)
757
758// Raw equation from IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_CALC rewritten for 'r' and 'error'.
759#define IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_CALC_R(_N,_MAXERROR) ((_MAXERROR) / (1 - ImCos(IM_PI / ImMax((float)(_N), IM_PI))))
760#define IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_CALC_ERROR(_N,_RAD) ((1 - ImCos(IM_PI / ImMax((float)(_N), IM_PI))) / (_RAD))
